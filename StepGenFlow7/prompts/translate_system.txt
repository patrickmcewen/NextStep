You are an expert at translating DSL-refactored PyTorch code into a complete STeP hardware graph.

You will receive DSL-refactored code where ALL operations are intermediate DSL function calls (from step_dsl.py). Each DSL call maps 1:1 to a STeP graph node. Your job: replace each DSL call with its STeP equivalent, wire them into a `Graph`, and return `(graph, output_op)`.

## Section 1: DSL → STeP IR Mapping

Each DSL function from step_dsl.py maps to a STeP graph node constructor. Source nodes (LinearOffChipLoad, SelectGen) do NOT take `graph` — you must call `graph.add_node(node)` manually. All other nodes take `graph` as the FIRST positional arg and auto-register.

Source nodes require `graph.add_node(node)`. All other nodes take `graph` as first arg and auto-register.

```python
class StepOps(ABC):
    _counter: int = 0
    instance_id: int

    def __init__(self):
        self.instance_id = StepOps._counter
        StepOps._counter += 1

    @property
    @abstractmethod
    def stream(self) -> Stream:
        """The stream of the operation."""
        pass

    @property
    @abstractmethod
    def stream_list(self) -> List[Stream]:
        pass

    @property
    @abstractmethod
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        pass

    @property
    @abstractmethod
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        pass

    @abstractmethod
    def stream_idx(self, idx: int) -> Stream:
        """The stream of the operation."""
        pass

    @abstractmethod
    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        pass

    @abstractmethod
    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        pass

    @abstractmethod
    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic (bytes) for this operation."""
        pass

    def is_offchip_memory_op(self) -> bool:
        """Override to True in off-chip memory operators."""
        return False

    @abstractmethod
    def T_fire(self, hw_config: dict) -> sympy.Expr:
        """Firing time: processing time per firing (Roofline). Returns cycles."""
        pass

    @abstractmethod
    def N_fire(self) -> sympy.Expr:
        """Firing count: total number of firings."""
        pass

    @abstractmethod
    def NIT(self) -> dict:
        """Input tile count per predecessor per firing.
        Returns {predecessor_instance_id: sympy.Expr}.
        For source nodes, returns empty dict."""
        pass

    @abstractmethod
    def OTPC(self) -> sympy.Expr:
        """Output tiles per chunk: output tiles produced per firing."""
        pass

class RandomOffChipLoad(StepOps):
    underlying: torch.Tensor
    tensor_shape_tiled: Tuple[int, ...]
    raddr: Union[StepOps, Tuple[StepOps, int]]
    tile_row: int
    tile_col: int
    n_byte: int
    base_addr_byte: int
    par_dispatch: int
    _stream: Stream
    transposed: bool
    track_traffic: bool

    def __init__(
        self,
        graph: MultiDiGraph,
        underlying: torch.Tensor,
        raddr: Union[StepOps, Tuple[StepOps, int]],
        tile_row: int,
        tile_col: int,
        base_addr_byte: int,
        par_dispatch: int,
        mock_bf16: bool = False,
        transposed: bool = False,
    ):
        super().__init__()

        self.underlying = underlying
        self.tensor_shape_tiled = tuple(
            list(underlying.shape[:-2])
            + [
                underlying.shape[-2] // tile_row,
                underlying.shape[-1] // tile_col,
            ]
        )
        self.raddr = raddr
        self.tile_row = tile_row
        self.tile_col = tile_col
        self.base_addr_byte = base_addr_byte
        self.par_dispatch = par_dispatch
        self.transposed = transposed

        if underlying.dtype == torch.float32:
            if mock_bf16:
                self.n_byte = 2  # we will use this to mimic bfloat16
            else:
                self.n_byte = 4

            stream_dtype = Tile(
                tile_dtype=Float32(),
                shape=(tile_row, tile_col) if not transposed else (tile_col, tile_row),
            )
            self._stream = Stream(
                stream_dtype=stream_dtype, shape=get_stream(raddr).shape
            )
        elif underlying.dtype == torch.float16:
            self.n_byte = 2

            stream_dtype = Tile(
                tile_dtype=Float16(),
                shape=(tile_row, tile_col) if not transposed else (tile_col, tile_row),
            )
            self._stream = Stream(
                stream_dtype=stream_dtype, shape=get_stream(raddr).shape
            )
        else:
            raise ValueError(f"Unsupported dtype: {underlying.dtype}")

        raddr_node = raddr if isinstance(raddr, StepOps) else raddr[0]
        graph.add_edge(raddr_node, self)

        self.track_traffic = False

    @property
    def stream(self) -> Stream:
        """The stream of the operation."""
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        return self.raddr

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return [self.raddr]

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id}"

    def replace_input(
        self,
        org_input: Union[StepOps, Tuple[StepOps, int]],
        new_input: Union[StepOps, Tuple[StepOps, int]],
    ):
        if get_stream(self.raddr) != get_stream(new_input):
            raise ValueError("The shape of the input stream shouldn't change")
        self.raddr = new_input

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        total_elements = self._stream.total_elements() * sympy.Integer(
            self.tile_row * self.tile_col * self.n_byte
        )
        return total_elements

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        return sympy.Integer(self.tile_row * self.tile_col * self.n_byte)

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        return sympy.Integer(0)

    def N_fire(self) -> sympy.Expr:
        return self._stream.total_elements()

    def NIT(self) -> dict:
        raddr_node = self.raddr if isinstance(self.raddr, StepOps) else self.raddr[0]
        return {raddr_node.instance_id: sympy.Integer(1)}

    def OTPC(self) -> sympy.Expr:
        return sympy.Integer(1)

    def is_offchip_memory_op(self) -> bool:
        return True


class RandomOffChipStore(StepOps):
    underlying: torch.Tensor
    tensor_shape_tiled: Tuple[int, ...]
    wdata: Union[StepOps, Tuple[StepOps, int]]
    waddr: Union[StepOps, Tuple[StepOps, int]]
    tile_row: int
    tile_col: int
    n_byte: int
    store_file_name: str
    ack_based_on_waddr: bool  # if true, the ack stream's shape will be based on the waddr, otherwise it is based on the wdata
    base_addr_byte: int
    buffer_depth: int
    par_dispatch: int
    _stream: Stream
    has_done_stream: bool

    def __init__(
        self,
        graph: MultiDiGraph,
        underlying: torch.Tensor,
        wdata: Union[StepOps, Tuple[StepOps, int]],
        waddr: Union[StepOps, Tuple[StepOps, int]],
        tile_row: int,
        tile_col: int,
        base_addr_byte: int,
        par_dispatch: int,
        ack_based_on_waddr: bool = True,
        buffer_depth: int = 1,
        mock_bf16: bool = False,
        has_done_stream: bool = False,
    ):
        super().__init__()

        self.underlying = underlying
        self.tensor_shape_tiled = tuple(
            list(underlying.shape[:-2])
            + [
                underlying.shape[-2] // tile_row,
                underlying.shape[-1] // tile_col,
            ]
        )
        self.wdata = wdata
        self.waddr = waddr
        self.tile_row = tile_row
        self.tile_col = tile_col
        self.base_addr_byte = base_addr_byte
        self.par_dispatch = par_dispatch
        self.buffer_depth = buffer_depth
        self.ack_based_on_waddr = ack_based_on_waddr
        self.store_file_name = f"{str(self)}.npy"
        self.has_done_stream = has_done_stream

        if underlying.dtype == torch.float32:
            if mock_bf16:
                self.n_byte = 2  # we will use this to mimic bfloat16
            else:
                self.n_byte = 4

            self._stream = Stream(
                stream_dtype=Bool(),
                shape=(
                    get_stream(waddr).shape
                    if ack_based_on_waddr
                    else get_stream(wdata).shape
                ),
            )
        elif underlying.dtype == torch.float16:
            self.n_byte = 2

            self._stream = Stream(
                stream_dtype=Bool(),
                shape=(
                    get_stream(waddr).shape
                    if ack_based_on_waddr
                    else get_stream(wdata).shape
                ),
            )
        else:
            raise ValueError(f"Unsupported dtype: {underlying.dtype}")

        waddr_node = waddr if isinstance(waddr, StepOps) else waddr[0]
        wdata_node = wdata if isinstance(wdata, StepOps) else wdata[0]
        graph.add_edge(waddr_node, self)
        graph.add_edge(wdata_node, self)

    @property
    def stream(self) -> Stream:
        """The stream of the operation."""
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        return NotImplementedError(
            "Shouldn't be called for nodes that has multiple inputs"
        )

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return [self.waddr, self.wdata]

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id}"

    def replace_input(
        self,
        org_input: Union[StepOps, Tuple[StepOps, int]],
        new_input: Union[StepOps, Tuple[StepOps, int]],
    ):
        if self.waddr == org_input:
            if get_stream(self.waddr) != get_stream(new_input):
                raise ValueError("The shape of the input stream shouldn't change")
            self.waddr = new_input
        elif self.wdata == org_input:
            if get_stream(self.wdata) != get_stream(new_input):
                raise ValueError("The shape of the input stream shouldn't change")
            self.wdata = new_input
        else:
            raise ValueError("Wrong org_input")

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        total_elements = self._stream.total_elements() * sympy.Integer(
            self.tile_row * self.tile_col * self.n_byte
        )
        return total_elements

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        return sympy.Integer(
            self.tile_row * self.tile_col * self.n_byte * self.buffer_depth
        )

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        return sympy.Integer(0)

    def N_fire(self) -> sympy.Expr:
        return self._stream.total_elements()

    def NIT(self) -> dict:
        waddr_node = self.waddr if isinstance(self.waddr, StepOps) else self.waddr[0]
        wdata_node = self.wdata if isinstance(self.wdata, StepOps) else self.wdata[0]
        return {waddr_node.instance_id: sympy.Integer(1), wdata_node.instance_id: sympy.Integer(1)}

    def OTPC(self) -> sympy.Expr:
        return sympy.Integer(1)

    def is_offchip_memory_op(self) -> bool:
        return True

    ref: Union[StepOps, Tuple[StepOps, int]]
    underlying: torch.Tensor
    tensor_shape_tiled: Tuple[int, ...]
    stride: Tuple[int, ...]
    out_shape_tiled: Tuple[int, ...]
    tile_row: int
    tile_col: int
    n_byte: int
    par_dispatch: int
    _stream: Stream
    transposed: bool
    trigger_rank: (
        int  # number of the inner dimensions that are used to trigger the load
    )

    def __init__(
        self,
        graph: MultiDiGraph,
        ref: Union[StepOps, Tuple[StepOps, int]],
        underlying: torch.Tensor,
        stride: Tuple[int, ...],
        out_shape_tiled: Tuple[int, ...],
        tile_row: int,
        tile_col: int,
        par_dispatch: int,
        mock_bf16: bool = False,
        transposed: bool = False,
        trigger_rank: int = 0,
    ):
        super().__init__()

        self.ref = ref
        self.underlying = underlying
        self.trigger_rank = trigger_rank
        self.tensor_shape_tiled = tuple(
            list(underlying.shape[:-2])
            + [
                underlying.shape[-2] // tile_row,
                underlying.shape[-1] // tile_col,
            ]
        )
        self.stride = stride
        self.out_shape_tiled = out_shape_tiled
        self.tile_row = tile_row
        self.tile_col = tile_col
        self.par_dispatch = par_dispatch
        self.transposed = transposed

        ref_node = ref if isinstance(ref, StepOps) else ref[0]
        graph.add_edge(ref_node, self)

        ref_stream: Stream = get_stream(ref)
        ref_stream_shape = (
            ref_stream.shape[: -self.trigger_rank]
            if self.trigger_rank > 0
            else ref_stream.shape
        )
        if underlying.dtype == torch.float32:
            if mock_bf16:
                self.n_byte = 2  # we will use this to mimic bfloat16
            else:
                self.n_byte = 4

            stream_dtype = Tile(
                tile_dtype=Float32(),
                shape=(tile_row, tile_col) if not transposed else (tile_col, tile_row),
            )
            self._stream = Stream(
                stream_dtype=stream_dtype, shape=ref_stream_shape + self.out_shape_tiled
            )
        elif underlying.dtype == torch.float16:
            self.n_byte = 2

            stream_dtype = Tile(
                tile_dtype=Float16(),
                shape=(tile_row, tile_col) if not transposed else (tile_col, tile_row),
            )
            self._stream = Stream(
                stream_dtype=stream_dtype, shape=ref_stream_shape + self.out_shape_tiled
            )
        else:
            raise ValueError(f"Unsupported dtype: {underlying.dtype}")

    @property
    def stream(self) -> Stream:
        """The stream of the operation."""
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        return self.ref

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return [self.ref]

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id} (rank={self.trigger_rank})"

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        if get_stream(self.ref) != get_stream(new_input):
            raise ValueError("The shape of the input stream shouldn't change")
        self.ref = new_input

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        total_elements = self._stream.total_elements() * sympy.Integer(
            self.tile_row * self.tile_col * self.n_byte
        )
        # print(f"LinearOffChipLoadRef: total_elements: {total_elements}")
        return total_elements

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        return sympy.Integer(self.tile_row * self.tile_col * self.n_byte)

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        return sympy.Integer(0)

    def N_fire(self) -> sympy.Expr:
        return self._stream.total_elements()

    def NIT(self) -> dict:
        ref_node = self.ref if isinstance(self.ref, StepOps) else self.ref[0]
        return {ref_node.instance_id: sympy.Integer(1)}

    def OTPC(self) -> sympy.Expr:
        # One trigger fires all out_shape_tiled tiles at HBM speed
        tiles_per_trigger = sympy.Integer(1)
        for d in self.out_shape_tiled:
            tiles_per_trigger = tiles_per_trigger * d
        return tiles_per_trigger

    def is_offchip_memory_op(self) -> bool:
        return True

class BinaryMap(StepOps):
    in1: Union[StepOps, Tuple[StepOps, int]]
    in2: Union[StepOps, Tuple[StepOps, int]]
    fn: MapFn
    write_back_mu: bool  # whether the consumer is a bufferize or not
    compute_bw: int
    _stream: Stream

    def __init__(
        self,
        graph: MultiDiGraph,
        in1: Union[StepOps, Tuple[StepOps, int]],
        in2: Union[StepOps, Tuple[StepOps, int]],
        fn: MapFn,
        write_back_mu: bool,
        compute_bw: int,
    ):

        super().__init__()

        self.in1 = in1
        self.in2 = in2
        self.fn = fn
        self.write_back_mu = write_back_mu
        self.compute_bw = compute_bw

        in1_stream: Stream = get_stream(in1)
        in2_stream: Stream = get_stream(in2)

        for dim in range(len(in1_stream.shape)):
            if isinstance(in1_stream.shape[dim], int):
                if in1_stream.shape[dim] != in2_stream.shape[dim]:
                    raise ValueError(
                        f"Input streams must have the same shape for the static dims: {in1_stream.shape} != {in2_stream.shape}"
                    )

        self._stream = Stream(
            stream_dtype=self.fn.apply(
                (in1_stream.stream_dtype, in2_stream.stream_dtype)
            ),
            shape=in1_stream.shape,
        )

        input_node1 = in1 if isinstance(in1, StepOps) else in1[0]
        input_node2 = in2 if isinstance(in2, StepOps) else in2[0]

        graph.add_edge(input_node1, self)
        graph.add_edge(input_node2, self)

    @property
    def stream(self) -> Stream:
        """The stream of the operation."""
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        return self.in1

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return [self.in1, self.in2]

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id} (fn: {self.fn})"

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        if self.in1 == org_input:
            if get_stream(self.in1) != get_stream(new_input):
                raise ValueError("The shape of the input stream shouldn't change")
            self.in1 = new_input
        elif self.in2 == org_input:
            if get_stream(self.in2) != get_stream(new_input):
                raise ValueError("The shape of the input stream shouldn't change")
            self.in2 = new_input
        else:
            raise ValueError("Wrong org_input")

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        return sympy.Integer(0)

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        if not count_fifos:
            return sympy.Integer(0)

        # Get streams for inputs
        in1_stream = get_stream(self.in1)
        in2_stream = get_stream(self.in2)

        # Calculate tile sizes in bytes for each stream
        assert not isinstance(
            in1_stream.stream_dtype, Buffer
        ), "Input stream must be a Tile type."
        assert not isinstance(
            in2_stream.stream_dtype, Buffer
        ), "Input stream must be a Tile type."
        assert not isinstance(
            self._stream.stream_dtype, Buffer
        ), "Input stream must be a Tile type."

        in1_stream_dtype_size = in1_stream.stream_dtype.size_in_bytes()
        in2_stream_dtype_size = in2_stream.stream_dtype.size_in_bytes()
        out_stream_dtype_size = self._stream.stream_dtype.size_in_bytes()

        return sympy.Add(
            in1_stream_dtype_size, in2_stream_dtype_size, out_stream_dtype_size
        )

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        in1_stream = get_stream(self.in1)
        in2_stream = get_stream(self.in2)
        flops = _compute_flops_per_tile(self.fn, [in1_stream, in2_stream], self._stream)
        out_tile_bytes = self._stream.stream_dtype.size_in_bytes()

class BinaryMapAccum(StepOps):
    in1: Union[StepOps, Tuple[StepOps, int]]
    in2: Union[StepOps, Tuple[StepOps, int]]
    fn: MapAccumFn
    init_fn: InitFn
    rank: int
    write_back_mu: bool  # whether the consumer is a bufferize or not
    compute_bw: int
    _stream: Stream

    # [Genghan] We need an init function?
    def __init__(
        self,
        graph: MultiDiGraph,
        in1: Union[StepOps, Tuple[StepOps, int]],
        in2: Union[StepOps, Tuple[StepOps, int]],
        fn: MapAccumFn,
        init_fn: InitFn,
        rank: int,
        write_back_mu: bool,
        compute_bw: int,
    ):

        super().__init__()

        self.in1 = in1
        self.in2 = in2
        self.fn = fn
        self.init_fn = init_fn
        self.rank = rank
        self.write_back_mu = write_back_mu
        self.compute_bw = compute_bw

        in1_stream: Stream = get_stream(in1)
        in2_stream: Stream = get_stream(in2)

        assert rank > 0, "Rank must be greater than 0."
        for in1_dim, in2_dim in zip(in1_stream.shape, in2_stream.shape):
            if isinstance(in1_dim, DynDim) and isinstance(in2_dim, DynDim):
                assert in1_dim.expr.equals(
                    in2_dim.expr
                ), f"Input streams must have the same shape. {in1_dim.expr} \n!=\n{in2_dim.expr}"
            else:
                assert (
                    in1_dim == in2_dim
                ), f"Input streams must have the same shape. {in1_dim} \n!=\n{in2_dim}"

        self._stream = Stream(
            stream_dtype=self.fn.apply(
                (in1_stream.stream_dtype, in2_stream.stream_dtype)
            ),
            shape=in1_stream.shape[: -self.rank],
        )

        input_node1 = in1 if isinstance(in1, StepOps) else in1[0]
        input_node2 = in2 if isinstance(in2, StepOps) else in2[0]

        graph.add_edge(input_node1, self)
        graph.add_edge(input_node2, self)

    @property
    def stream(self) -> Stream:
        """The stream of the operation."""
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        return self.in1

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return [self.in1, self.in2]

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id} (fn: {self.fn})"

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        if self.in1 == org_input:
            if get_stream(self.in1) != get_stream(new_input):
                raise ValueError("The shape of the input stream shouldn't change")
            self.in1 = new_input
        elif self.in2 == org_input:
            if get_stream(self.in2) != get_stream(new_input):
                raise ValueError("The shape of the input stream shouldn't change")
            self.in2 = new_input
        else:
            raise ValueError("Wrong org_input")

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        return sympy.Integer(0)

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        if not count_fifos:
            # Get the size of the stream's data type
            return self._stream.stream_dtype.size_in_bytes()

        # Get the first input stream (all inputs should have the same datatype)
        in_stream = get_stream(self.in1)

        # Check that the input stream's datatype is a Tile
        assert isinstance(
            in_stream.stream_dtype, Tile
        ), "Input stream must be a Tile type."

        # Calculate the size of the input stream's datatype
        in_tile_size = in_stream.stream_dtype.size_in_bytes()

        # Return the size times (len(self._inputs) + 1) for FIFO requirements
        return sympy.Mul(in_tile_size, sympy.Integer(len(self.input_list) + 1))

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        in1_stream = get_stream(self.in1)
        in2_stream = get_stream(self.in2)
        flops = _compute_flops_per_tile(self.fn, [in1_stream, in2_stream], self._stream)
        out_tile_bytes = self._stream.stream_dtype.size_in_bytes()
        return _compute_roofline(flops, self.compute_bw, out_tile_bytes, self.write_back_mu)

    def N_fire(self) -> sympy.Expr:
        return self._stream.total_elements()

    def NIT(self) -> dict:
        in1_stream = get_stream(self.in1)
        k = sympy.Integer(1)
        for dim in in1_stream.shape[-self.rank:]:
            k = k * (dim.expr if isinstance(dim, DynDim) else dim)
        in1_node = self.in1 if isinstance(self.in1, StepOps) else self.in1[0]
        in2_node = self.in2 if isinstance(self.in2, StepOps) else self.in2[0]
        return {in1_node.instance_id: k, in2_node.instance_id: k}

    def OTPC(self) -> sympy.Expr:
        return sympy.Integer(1)

        else:

                self.tensor_shape_tiled[-2] * self.tile_row,
                self.tensor_shape_tiled[-1] * self.tile_col,
            )

                f"The shape of the input stream shouldn't change ({get_stream(self.input)} != {get_stream(new_input)})"

        """Return the on-chip memory requirement for this operation."""
        # Get the input stream
        input_stream = get_stream(self._input)

        # Return the size of the data type of the input stream
        return input_stream.stream_dtype.size_in_bytes()

        return sympy.Integer(0)

        input_node = self._input if isinstance(self._input, StepOps) else self._input[0]
        return {input_node.instance_id: sympy.Integer(1)}

class Bufferize(StepOps):

        new_input: Union["StepOps", Tuple["StepOps", int]],

        if get_stream(self.input) != get_stream(new_input):
            raise ValueError("The shape of the input stream shouldn't change")
        self._input = new_input

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        if not self.off_chip:
            return sympy.Integer(0)

        # Get the input stream
        in_stream = get_stream(self._input)

        # Check that the input stream's datatype is a Tile
        assert isinstance(
            in_stream.stream_dtype, Tile
        ), "Input stream must be a Tile type."
        tile_size = in_stream.stream_dtype.size_in_bytes()

        # Calculate the total number of elements in the input stream
        total_elements = in_stream.total_elements()

        return sympy.Mul(total_elements, tile_size)

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:

            self.off_chip

            self._stream.stream_dtype, Buffer

        return sympy.Integer(1)



class Streamify(StepOps):
    _input: Union[StepOps, Tuple[StepOps, int]]
    repeat_factor: List[int]
    rank: int
    off_chip: bool
    _stream: Stream

    def __init__(
        self,
        graph: MultiDiGraph,
        input: Union[StepOps, Tuple[StepOps, int]],
        repeat_factor: List[int],
        rank: int,  # The rank of the Buffer
        off_chip: bool = False,
    ):
        super().__init__()

        self._input = input
        self.repeat_factor = repeat_factor
        self.rank = rank
        self.off_chip = off_chip

        in_stream: Stream = get_stream(input)
        assert rank > 0, "Rank must be greater than 0."
        assert isinstance(
            in_stream.stream_dtype, Buffer
        ), "Input stream must be a Buffer type."

        buffer_shape = in_stream.stream_dtype.shape
        buffer_dtype = in_stream.stream_dtype.buff_dtype
        self._stream = Stream(
            stream_dtype=buffer_dtype,
            shape=in_stream.shape + tuple(repeat_factor) + buffer_shape,
        )

        input_node = input if isinstance(input, StepOps) else input[0]
        graph.add_edge(input_node, self)

    @property
    def stream(self) -> Stream:
        """The stream of the operation."""
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        return self._input

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return [self._input]

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id} ({self.rank} D)"

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        if get_stream(self.input) != get_stream(new_input):
            raise ValueError("The shape of the input stream shouldn't change")
        self._input = new_input

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        if not self.off_chip:
            return sympy.Integer(0)

        # Get the input stream
        in_stream = get_stream(self._input)
        assert isinstance(
            in_stream.stream_dtype, Buffer
        ), "Input stream must be a Buffer type."
        buffer_size = self._stream.stream_dtype.size_in_bytes()

        # Calculate the total number of elements in the input stream
        total_input_elements = in_stream.total_elements()

        # Calculate the product of elements in repeat_factor
        repeat_product = sympy.Mul(*self.repeat_factor)

        return sympy.Mul(total_input_elements, repeat_product, buffer_size)

    def on_chip_requirement(self, count_fifos: bool = True) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        # Get the size of the stream's data type
        assert isinstance(
            self._stream.stream_dtype, Tile
        ), "Stream must be a Tile type."
        tile_size = self._stream.stream_dtype.size_in_bytes()

        if self.off_chip:
            return tile_size

        in_stream = get_stream(self._input)
        assert isinstance(
            in_stream.stream_dtype, Buffer
        ), "Stream must be a Buffer type."
        # Calculate the size of the buffer (self._stream's datatype)
        buffer_size = in_stream.stream_dtype.size_in_bytes()

        return sympy.Add(buffer_size, tile_size)

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        # Simulator does 1 cycle per output element (tile)
        return self.OTPC()

    def N_fire(self) -> sympy.Expr:
        return self._stream.total_elements()

    def NIT(self) -> dict:
        input_node = self._input if isinstance(self._input, StepOps) else self._input[0]
        return {input_node.instance_id: sympy.Integer(1)}

    def OTPC(self) -> sympy.Expr:
        in_stream = get_stream(self._input)
        assert isinstance(in_stream.stream_dtype, Buffer)
        e = sympy.Integer(1)
        for dim in in_stream.stream_dtype.shape:
            e = e * (dim.expr if isinstance(dim, DynDim) else dim)
        for rf in self.repeat_factor:
            e = e * rf
        return e



class DynStreamify(StepOps):
    _input: Union[StepOps, Tuple[StepOps, int]]
    ref: Union[StepOps, Tuple[StepOps, int]]
    repeat_rank: int
    bufferized_rank: int
    _stream: Stream
    off_chip: bool

    # [Genghan] There is an ExpandRef hidden in the operation
    def __init__(
        self,
        graph: MultiDiGraph,
        input: Union[StepOps, Tuple[StepOps, int]],
        ref: Union[StepOps, Tuple[StepOps, int]],
        repeat_rank: int,  # Starting from this rank to rank 0, the input_stream should have 1s
        bufferized_rank: int,
        off_chip: bool = False,
    ):
        super().__init__()

        self._input = input
        self.ref = ref
        self.repeat_rank = repeat_rank
        self.bufferized_rank = bufferized_rank
        self.off_chip = off_chip

        in_stream: Stream = get_stream(input)
        ref_stream: Stream = get_stream(ref)

        assert bufferized_rank > 0, "Bufferized rank must be greater than 0."
        calc_rank = self.repeat_rank + 1
        assert (
            in_stream.shape[:-calc_rank] == ref_stream.shape[:-calc_rank]
        ), f"Shapes up to the repeat rank don't match: {in_stream.shape[: -calc_rank]} != {ref_stream.shape[: -calc_rank]}"

        assert (
            in_stream.shape[-calc_rank:] == (1,) * calc_rank
        ), f"Input stream shape must have 1s in the repeat rank dimensions {in_stream.shape[-calc_rank :]} != {(1,) * calc_rank}."

        assert isinstance(
            in_stream.stream_dtype, Buffer
        ), "Input stream must be a Buffer type."

        buffer_shape = in_stream.stream_dtype.shape
        buffer_dtype = in_stream.stream_dtype.buff_dtype

        self._stream = Stream(
            stream_dtype=buffer_dtype, shape=ref_stream.shape + buffer_shape
        )

        input_node = input if isinstance(input, StepOps) else input[0]
        graph.add_edge(input_node, self)

        ref_node = ref if isinstance(ref, StepOps) else ref[0]
        graph.add_edge(ref_node, self)

    @property
    def stream(self) -> Stream:
        """The stream of the operation."""
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        return self._input

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return [self._input, self.ref]

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id}"

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        if self._input == org_input:
            if get_stream(self._input) != get_stream(new_input):
                raise ValueError("The shape of the input stream shouldn't change")
            self._input = new_input
        elif self.ref == org_input:
            if get_stream(self.ref) != get_stream(new_input):
                raise ValueError("The shape of the ref stream shouldn't change")
            self.ref = new_input
        else:
            raise ValueError("Wrong org_input")

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        if not self.off_chip:
            return sympy.Integer(0)

        # Get the ref stream
        ref_stream = get_stream(self.ref)

        # Calculate the product of elements in the ref stream's shape
        ref_elements = ref_stream.total_elements()

        # Get the size of the buffer (the data type of self._stream)
        buffer_size = self._stream.stream_dtype.size_in_bytes()

        return sympy.Mul(ref_elements, buffer_size)

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        # Get the size of the stream's data type
        assert isinstance(
            self._stream.stream_dtype, Tile
        ), "Stream must be a Tile type."
        tile_size = self._stream.stream_dtype.size_in_bytes()

        if self.off_chip:
            return tile_size

        in_stream = get_stream(self._input)
        assert isinstance(
            in_stream.stream_dtype, Buffer
        ), "Stream must be a Buffer type."
        # Calculate the size of the buffer (self._stream's datatype)
        buffer_size = in_stream.stream_dtype.size_in_bytes()

        return sympy.Add(buffer_size, tile_size)

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        tile_size_bits = self._stream.stream_dtype.size_in_bytes() * 8
        scratchpad_bw = hw_config.get("scratchpad_bw", 1024)
        return sympy.ceiling(tile_size_bits / scratchpad_bw)

    def N_fire(self) -> sympy.Expr:
        ref_stream = get_stream(self.ref)
        return ref_stream.total_elements()

    def NIT(self) -> dict:
        input_node = self._input if isinstance(self._input, StepOps) else self._input[0]
        ref_node = self.ref if isinstance(self.ref, StepOps) else self.ref[0]
        return {input_node.instance_id: sympy.Integer(1), ref_node.instance_id: sympy.Integer(1)}

    def OTPC(self) -> sympy.Expr:
        in_stream = get_stream(self._input)
        assert isinstance(in_stream.stream_dtype, Buffer)
        e = sympy.Integer(1)
        for dim in in_stream.stream_dtype.shape:
            e = e * (dim.expr if isinstance(dim, DynDim) else dim)
        return e



class Parallelize(StepOps):
    """
    [Da,...,D0] x 1 -> [Da//par, ..., D0] x par
    b = parallelize rank
    """

    _input: Union[StepOps, Tuple[StepOps, int]]
    parallelize_rank: int
    num_consumers: int
    switch_cycles: List[int]
    write_back_mu: bool
    _stream: List[Stream]

    def __init__(
        self,
        graph: MultiDiGraph,
        input: Union[StepOps, Tuple[StepOps, int]],
        parallelize_rank: int,
        num_consumers: int,  # this is the par factor
        switch_cycles: List[int] = None,
        write_back_mu: bool = False,
    ):
        super().__init__()
        self._input = input
        self.parallelize_rank = parallelize_rank
        self.num_consumers = num_consumers
        self.switch_cycles = (
            switch_cycles if switch_cycles is not None else [1] * num_consumers
        )
        self.write_back_mu = write_back_mu

        in_stream: Stream = get_stream(input)
        assert (
            parallelize_rank == in_stream.rank
        ), "Parallelize rank must be the same as the rank of the input stream"

        if isinstance(in_stream.shape[0], DynDim):
            # As the cases for use, we sum all the parallel regions, we simply do a truediv
            self._stream = [
                Stream(
                    stream_dtype=in_stream.stream_dtype,
                    shape=(in_stream.shape[0] / num_consumers,) + in_stream.shape[1:],
                )
                for i in range(num_consumers)
            ]

        else:
            if in_stream.shape[0] % (num_consumers) != 0:
                raise NotImplementedError(
                    "The symbolic shape for (in_stream.shape[0] % (num_consumers) != 0) is not supported yet "
                )
            else:
                self._stream = [
                    Stream(
                        stream_dtype=in_stream.stream_dtype,
                        shape=(in_stream.shape[0] // num_consumers,)
                        + in_stream.shape[1:],
                    )
                    for i in range(num_consumers)
                ]

        input_node = input if isinstance(input, StepOps) else input[0]
        graph.add_edge(input_node, self)

    @property
    def stream(self) -> Stream:
        raise NotImplementedError(
            "This property shouldn't be used for nodes with multiple output streams"
        )

    @property
    def stream_list(self) -> List[Stream]:
        return self._stream

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        return self._input

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return [self._input]

    def stream_idx(self, idx: int) -> Stream:
        return self._stream[idx]

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id} ({self.parallelize_rank} D)"

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        if self.input == org_input:
            if get_stream(self.input) != get_stream(new_input):
                raise ValueError("The shape of the input stream shouldn't change")
            self._input = new_input
        else:
            raise ValueError("Wrong org_input")

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        return sympy.Integer(0)

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        if not count_fifos:
            return sympy.Integer(0)

        # Get the input stream
        in_stream = get_stream(self._input)

        # Check that the input stream's datatype is a Tile
        assert isinstance(
            in_stream.stream_dtype, (Tile, Select)
        ), "Input stream must be a Tile or Select type."

        # Calculate the size of the input stream's datatype
        in_tile_size = in_stream.stream_dtype.size_in_bytes()

        # Return the size times (num_consumers + 1) for FIFO requirements
        return sympy.Mul(in_tile_size, sympy.Integer(self.num_consumers + 1))

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        return sympy.Integer(0)

    def N_fire(self) -> sympy.Expr:
        total = sympy.Integer(0)
        for s in self._stream:
            total = total + s.total_elements()
        return total

    def NIT(self) -> dict:
        input_node = self._input if isinstance(self._input, StepOps) else self._input[0]
        return {input_node.instance_id: sympy.Integer(1)}

    def OTPC(self) -> sympy.Expr:
        return sympy.Integer(1)



class StaticReassemble(StepOps):
    """
    Inverse of Parallelize.
    [Da//par, ..., D0] x par -> [Da,...,D0] x 1
    Merges multiple parallel streams back into a single stream.
    """

    _inputs: List[Union[StepOps, Tuple[StepOps, int]]]
    merge_rank: int
    switch_cycles: List[int]
    write_back_mu: bool
    _stream: Stream

    def __init__(
        self,
        graph: MultiDiGraph,
        inputs: List[Union[StepOps, Tuple[StepOps, int]]],
        merge_rank: int,
        switch_cycles: List[int] = None,
        write_back_mu: bool = False,
    ):
        super().__init__()

        self._inputs = inputs
        self.merge_rank = merge_rank
        self.switch_cycles = (
            switch_cycles if switch_cycles is not None else [1] * len(inputs)
        )
        self.write_back_mu = write_back_mu

        in_streams: List[Stream] = [get_stream(input) for input in inputs]

        self._stream = Stream(
            stream_dtype=in_streams[0].stream_dtype,
            shape=(in_streams[0].shape[0] * len(in_streams),) + in_streams[0].shape[1:],
        )

        for input_node in inputs:
            node = input_node if isinstance(input_node, StepOps) else input_node[0]
            graph.add_edge(node, self)

    @property
    def stream(self) -> Stream:
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        raise NotImplementedError(
            "Shouldn't be called for nodes that has multiple input streams"
        )

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return self._inputs

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id} ({self.merge_rank} D)"

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        for i, input_node in enumerate(self._inputs):
            if input_node == org_input:
                if get_stream(input_node) != get_stream(new_input):
                    raise ValueError("The shape of the input stream shouldn't change")
                self._inputs[i] = new_input
                return
        raise ValueError("Wrong org_input")

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        return sympy.Integer(0)

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        if not count_fifos:
            return sympy.Integer(0)

        # Get the first input stream (all inputs should have the same datatype)
        in_stream = get_stream(self._inputs[0])

        # Check that the input stream's datatype is a Tile or Select
        assert isinstance(
            in_stream.stream_dtype, (Tile, Select)
        ), "Input stream must be a Tile or Select type."

        # Calculate the size of the input stream's datatype
        in_tile_size = in_stream.stream_dtype.size_in_bytes()

        # Return the size times (len(self._inputs) + 1) for FIFO requirements
        return sympy.Mul(in_tile_size, sympy.Integer(len(self._inputs) + 1))

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        return sympy.Integer(0)

    def N_fire(self) -> sympy.Expr:
        return self._stream.total_elements()

    def NIT(self) -> dict:
        result = {}
        for inp in self._inputs:
            node = inp if isinstance(inp, StepOps) else inp[0]
            result[node.instance_id] = sympy.Integer(1)
        return result

    def OTPC(self) -> sympy.Expr:
        return sympy.Integer(1)

class EagerMerge(StepOps):
    _inputs: List[Union[StepOps, Tuple[StepOps, int]]]
    input_rank: int  # this is the merge (reassemble) rank too
    num_consumers: int
    _stream: List[Stream]

    def __init__(
        self,
        graph: MultiDiGraph,
        inputs: List[Union[StepOps, Tuple[StepOps, int]]],
        input_rank: int,  # Remove dimensions at rank larger or equal to this value
    ):
        super().__init__()

        self._inputs = inputs
        self.input_rank = input_rank
        self.num_consumers = 2

        in_streams: List[Stream] = [get_stream(input) for input in inputs]
        # assert all(
        #     stream.shape[1:] == in_streams[0].shape[1:] for stream in in_streams
        # ), "All input streams must have the same shape except the outermost dimensions."

        # True if any input streams has a dynamic outermost dimension
        has_dyndim_input = any(
            isinstance(stream.shape[0], DynDim) for stream in in_streams
        )
        merged_dim = DynDim(sympy.Integer(0)) if has_dyndim_input else 0
        merged_dim = sum([stream.shape[0] for stream in in_streams], start=merged_dim)

        stream_dtype = in_streams[0].stream_dtype
        if not all(stream.stream_dtype == stream_dtype for stream in in_streams):
            if all(isinstance(stream.stream_dtype, DynTile) for stream in in_streams):
                # Helper to compute max that works with both int and DynDim
                def symbolic_max(values):
                    # Check if any value is DynDim
                    has_dyndim = any(isinstance(v, DynDim) for v in values)
                    if has_dyndim:
                        # Convert all to sympy expressions
                        exprs = [
                            v.expr if isinstance(v, DynDim) else sympy.Integer(v)
                            for v in values
                        ]
                        return DynDim(sympy.Max(*exprs))
                    else:
                        # All are integers, use regular max
                        return max(values)

                tile_row = symbolic_max(
                    [row_i.stream_dtype.shape[0] for row_i in in_streams]
                )
                tile_col = symbolic_max(
                    [col_i.stream_dtype.shape[1] for col_i in in_streams]
                )

                stream_dtype = DynTile(
                    tile_dtype=stream_dtype.tile_dtype,
                    shape=(tile_row, tile_col),
                )

            else:
                raise ValueError(
                    "All input streams must have the same data type or be DynTile types."
                )
        data_stream = Stream(
            stream_dtype=stream_dtype,
            # stream_dtype=in_streams[0].stream_dtype,
            shape=(merged_dim,) + in_streams[0].shape[1:],
        )
        sel_stream = Stream(
            stream_dtype=MultiHot(total_n=len(in_streams)),
            shape=(merged_dim,),
        )
        self._stream = [data_stream, sel_stream]

        for input_node in inputs:
            node = input_node if isinstance(input_node, StepOps) else input_node[0]
            graph.add_edge(node, self)

    @property
    def stream(self) -> Stream:
        raise NotImplementedError(
            "This property shouldn't be used for nodes with multiple output streams"
        )

    @property
    def stream_list(self) -> List[Stream]:
        return self._stream

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        raise NotImplementedError(
            "Shouldn't be called for nodes that has multiple input streams"
        )

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return self._inputs

    def stream_idx(self, idx: int) -> Stream:
        """
        idx:
        0: data stream
        1: select stream
        """
        assert idx in [0, 1], "Invalid stream index"
        return self._stream[idx]

    def data_stream(self) -> Stream:
        return self._stream[0]

    def data_tuple(self) -> Tuple[StepOps, int]:
        return (self, 0)

    def select_tuple(self) -> Tuple[StepOps, int]:
        return (self, 1)

    def select_stream(self) -> Stream:
        return self._stream[1]

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id} ({self.input_rank} D)"

    def replace_full_input(
        self,
        graph: MultiDiGraph,
        new_input_list: List[Union["StepOps", Tuple["StepOps", int]]],
    ):
        assert len(self._inputs) == len(
            new_input_list
        ), "The number of inputs should be the same"

        for org_node, new_node in zip(self._inputs, new_input_list):
            if get_stream(org_node).shape != get_stream(new_node).shape:
                raise ValueError("The shape of the input stream shouldn't change")

        for input_node in self._inputs:
            node = input_node if isinstance(input_node, StepOps) else input_node[0]
            graph.remove_node(node)

        self._inputs = new_input_list

        for input_node in new_input_list:
            node = input_node if isinstance(input_node, StepOps) else input_node[0]
            graph.add_edge(node, self)

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        for i, input_node in enumerate(self._inputs):
            if input_node == org_input:
                if get_stream(input_node) != get_stream(new_input):
                    raise ValueError("The shape of the input stream shouldn't change")
                self._inputs[i] = new_input
                return

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        return sympy.Integer(0)

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        if not count_fifos:
            return sympy.Integer(0)

        # Get the first input stream (all inputs should have the same datatype)
        in_stream = get_stream(self._inputs[0])

        sel_stream_out: Stream = self.stream_idx(1)

        # Check that the input stream's datatype is a Tile
        assert isinstance(
            in_stream.stream_dtype, Tile
        ), "Input stream must be a Tile type."

        # Calculate the size of the input stream's datatype
        in_tile_size = in_stream.stream_dtype.size_in_bytes()
        sel_stream_out_size = sel_stream_out.stream_dtype.size_in_bytes()

        # Return the size times (len(self._inputs) + 1) for FIFO requirements

        return sympy.Mul(
            sympy.Add(in_tile_size, sel_stream_out_size),
            sympy.Integer(len(self._inputs) + 1),
        )

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        return sympy.Integer(0)

    def N_fire(self) -> sympy.Expr:
        return self._stream[0].total_elements()

    def NIT(self) -> dict:
        result = {}
        for inp in self._inputs:
            node = inp if isinstance(inp, StepOps) else inp[0]
            result[node.instance_id] = sympy.Integer(1)
        return result

    def OTPC(self) -> sympy.Expr:
        return sympy.Integer(1)



class FlatReassemble(StepOps):
    _inputs: List[Union[StepOps, Tuple[StepOps, int]]]
    control: Union[StepOps, Tuple[StepOps, int]]
    reassemble_rank: int
    switch_cycles: List[int]
    write_back_mu: bool
    _stream: Stream

    def __init__(
        self,
        graph: MultiDiGraph,
        inputs: List[Union[StepOps, Tuple[StepOps, int]]],
        control: Union[StepOps, Tuple[StepOps, int]],
        reassemble_rank: int,  # Remove dimensions at rank larger or equal to this value
        switch_cycles: List[int],
        write_back_mu: bool,
    ):
        super().__init__()

        self._inputs = inputs
        self.control = control
        self.reassemble_rank = reassemble_rank
        self.switch_cycles = switch_cycles
        self.write_back_mu = write_back_mu

        in_streams = [get_stream(input) for input in inputs]
        assert all(
            stream.shape[len(stream.shape) - reassemble_rank :]
            == in_streams[0].shape[len(in_streams[0].shape) - reassemble_rank :]
            for stream in in_streams
        ), "All input streams must have the same shape for the last 'reassemble_rank' dimensions."
        control_stream: Stream = get_stream(control)
        new_name = DynDim(f"{str(self)}_dyn")
        self._stream = Stream(
            stream_dtype=in_streams[0].stream_dtype,
            shape=control_stream.shape
            + (new_name,)
            + in_streams[0].shape[len(in_streams[0].shape) - reassemble_rank :],
        )

        for input_node in inputs:
            node = input_node if isinstance(input_node, StepOps) else input_node[0]
            graph.add_edge(node, self)

        control_node = control if isinstance(control, StepOps) else control[0]
        graph.add_edge(control_node, self)

    @property
    def stream(self) -> Stream:
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        raise NotImplementedError(
            "Shouldn't be called for nodes that has multiple input streams"
        )

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return self._inputs + [self.control]

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id} ({self.reassemble_rank} D)"

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        for i, input_node in enumerate(self._inputs):
            if input_node == org_input:
                if get_stream(input_node) != get_stream(new_input):
                    raise ValueError("The shape of the input stream shouldn't change")
                self._inputs[i] = new_input
                return

        if self.control == org_input:
            if get_stream(self.control) != get_stream(new_input):
                raise ValueError("The shape of the input stream shouldn't change")
            self.control = new_input
        else:
            raise ValueError("Wrong org_input")

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        return sympy.Integer(0)

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        if not count_fifos:
            if self.write_back_mu:
                reassembled_elements = sympy.Integer(1)
                for dim in self._stream.shape[get_stream(self.control).rank - 1 :]:
                    if isinstance(dim, DynDim):
                        reassembled_elements *= dim.expr
                    else:
                        reassembled_elements *= dim
                return reassembled_elements * self._stream.stream_dtype.size_in_bytes()
            return sympy.Integer(0)

        # Get the first input stream (all inputs should have the same datatype)
        in_stream = get_stream(self._inputs[0])

        # Check that the input stream's datatype is a Tile
        assert isinstance(
            in_stream.stream_dtype, Tile
        ), "Input stream must be a Tile type."

        # Calculate the size of the input stream's datatype
        in_tile_size = in_stream.stream_dtype.size_in_bytes()

        if self.write_back_mu:
            reassembled_elements = sympy.Integer(1)
            for dim in self._stream.shape[get_stream(self.control).rank - 1 :]:
                if isinstance(dim, DynDim):
                    reassembled_elements *= dim.expr
                else:
                    reassembled_elements *= dim
            return sympy.Add(
                sympy.Mul(in_tile_size, sympy.Integer(len(self._inputs) + 1)),
                reassembled_elements * self._stream.stream_dtype.size_in_bytes(),
            )

        # Return the size times (len(self._inputs) + 1) for FIFO requirements

        return sympy.Mul(in_tile_size, sympy.Integer(len(self._inputs) + 1))

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        return sympy.Integer(0)

    def N_fire(self) -> sympy.Expr:
        return self._stream.total_elements()

    def NIT(self) -> dict:
        result = {}
        for inp in self._inputs:
            node = inp if isinstance(inp, StepOps) else inp[0]
            result[node.instance_id] = sympy.Integer(1)
        ctrl_node = self.control if isinstance(self.control, StepOps) else self.control[0]
        result[ctrl_node.instance_id] = sympy.Integer(1)
        return result

    def OTPC(self) -> sympy.Expr:
        return sympy.Integer(1)

class Accum(StepOps):

    fn: AccumFn
    init_fn: InitFn
    accum_rank: int
    write_back_mu: bool
    compute_bw: int
    _stream: Stream

    def __init__(
        self,
        graph: MultiDiGraph,

        output_stream_dtype: Union[DynTile, Tile, Buffer, Select],
        fn: AccumFn,
        init_fn: InitFn,
        accum_rank: int,
        write_back_mu: bool,
        compute_bw: int,
    ):
        super().__init__()

        self._input = input
        self.fn = fn
        self.init_fn = init_fn
        self.accum_rank = accum_rank
        self.write_back_mu = write_back_mu
        self.compute_bw = compute_bw

        in_stream: Stream = get_stream(input)
        assert accum_rank > 0, "Accum rank must be greater than 0."

        self._stream = Stream(
            stream_dtype=output_stream_dtype,
            shape=in_stream.shape[: -self.accum_rank],
        )

        input_node = input if isinstance(input, StepOps) else input[0]
        graph.add_edge(input_node, self)

    @property
    def stream(self) -> Stream:
        """The stream of the operation."""
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        return self._input

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return [self._input]

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id} ({self.accum_rank} D) (fn: {self.fn})"

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        if get_stream(self._input) != get_stream(new_input):
            raise ValueError("The shape of the input stream shouldn't change")
        self._input = new_input

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        return sympy.Integer(0)

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        accumulator_size = self._stream.stream_dtype.size_in_bytes()

        in_stream_dtype = get_stream(self._input).stream_dtype
        assert not isinstance(
            in_stream_dtype, Buffer
        ), "Input stream must be a Tile type."
        input_size = in_stream_dtype.size_in_bytes()
        if not count_fifos:
            return accumulator_size

        return sympy.Add(accumulator_size, input_size)

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        in_stream = get_stream(self._input)
        in_tile = in_stream.stream_dtype
        # FLOPs per accumulation step: 1 op per input element
        if isinstance(in_tile, (Tile, DynTile)):
            r, c = in_tile.shape[0], in_tile.shape[1]
            flops = r * c if isinstance(r, int) and isinstance(c, int) else r * c
        else:
            flops = sympy.Integer(1)
        out_tile_bytes = self._stream.stream_dtype.size_in_bytes()
        return _compute_roofline(sympy.Integer(flops), self.compute_bw, out_tile_bytes, self.write_back_mu)

    def N_fire(self) -> sympy.Expr:
        return self._stream.total_elements()

    def NIT(self) -> dict:
        in_stream = get_stream(self._input)
        k = sympy.Integer(1)
        for dim in in_stream.shape[-self.accum_rank:]:
            k = k * (dim.expr if isinstance(dim, DynDim) else dim)
        input_node = self._input if isinstance(self._input, StepOps) else self._input[0]
        return {input_node.instance_id: k}

    def OTPC(self) -> sympy.Expr:
        return sympy.Integer(1)



class Flatten(StepOps):
    _input: Union[StepOps, Tuple[StepOps, int]]
    min_rank: int
    max_rank: int
    _stream: Stream

    def __init__(
        self,
        graph: MultiDiGraph,
        input: Union[StepOps, Tuple[StepOps, int]],
        min_rank: int,
        max_rank: int,
    ):
        super().__init__()
        self._input = input
        self.min_rank = min_rank
        self.max_rank = max_rank

        input_stream: Stream = get_stream(input)
        self._stream = Stream(
            stream_dtype=input_stream.stream_dtype,
            shape=tuple(
                self._compute_flattened_shape(input_stream.shape, min_rank, max_rank)
            ),
        )

        input_node = input if isinstance(input, StepOps) else input[0]
        graph.add_edge(input_node, self)

    def _compute_flattened_shape(
        self, shape: Tuple[int | DynDim, ...], min_rank, max_rank
    ):
        # Convert ranks to indices (rank 0 = rightmost = highest index)
        min_index = len(shape) - 1 - max_rank  # Note: max_rank gives min_index
        max_index = len(shape) - 1 - min_rank  # Note: min_rank gives max_index

        # Validate indices
        if min_index < 0 or max_index >= len(shape) or min_index > max_index:
            raise ValueError("Invalid rank range")

        # Calculate merged dimension
        merged_dim: Union[int, DynDim] = 1
        for i in range(min_index, max_index + 1):
            if isinstance(shape[i], DynDim):
                merged_dim = shape[i] * merged_dim  # type: ignore
            else:
                merged_dim = merged_dim * shape[i]  # type: ignore

        # Build new shape
        new_shape = shape[:min_index] + (merged_dim,) + shape[max_index + 1 :]

        return new_shape

    @property
    def stream(self) -> Stream:
        """The stream of the operation."""
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        return self._input

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return [self._input]

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id} ({self.min_rank} D, {self.max_rank} D)"

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        if get_stream(self.input) != get_stream(new_input):
            raise ValueError("The shape of the input stream shouldn't change")
        self._input = new_input

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        return sympy.Integer(0)

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        if not count_fifos:
            return sympy.Integer(0)

        # Get the size of the stream's data type
        stream_size = self._stream.stream_dtype.size_in_bytes()
        return sympy.Mul(stream_size, sympy.Integer(2))

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        return sympy.Integer(0)

    def N_fire(self) -> sympy.Expr:
        return self._stream.total_elements()

    def NIT(self) -> dict:
        input_node = self._input if isinstance(self._input, StepOps) else self._input[0]
        return {input_node.instance_id: sympy.Integer(1)}

    def OTPC(self) -> sympy.Expr:
        return sympy.Integer(1)

class Reshape(StepOps):
    _input: Union[StepOps, Tuple[StepOps, int]]
    chunk_size: int
    reshape_rank: int
    pad_fn: Optional[InitFn]
    write_back_mu: bool
    input_stream_rank: int
    add_outer_dim: bool
    _stream: Stream

    def __init__(
        self,
        graph: MultiDiGraph,
        input: Union[StepOps, Tuple[StepOps, int]],
        chunk_size: int,
        reshape_rank: int,
        write_back_mu: bool,
        add_outer_dim: bool = False,
        pad_fn: Optional[InitFn] = None,
    ):
        super().__init__()
        self._input = input
        self.chunk_size = chunk_size
        self.reshape_rank = reshape_rank
        self.pad_fn = pad_fn
        self.write_back_mu = write_back_mu

        in_stream: Stream = get_stream(input)

        ), "The chunk size must be a divisor of the shape at the reshape rank if the rank being split is not the innermost"

        if add_outer_dim:
            assert (
                in_stream.rank == 0
            ), "Input stream rank must be 0 if add_outer_dim is True"
        self.add_outer_dim = add_outer_dim

        assert (
            reshape_rank >= 0 and reshape_rank <= in_stream.rank
        ), f"Reshape rank must be between 0 and {in_stream.rank}."

        if isinstance(in_stream.shape[rank_pos], DynDim):
            if add_outer_dim:  # this means in_stream.rank == 0
                outer_most_expr: sympy.Expr = in_stream.shape[rank_pos].expr
                # print(f"outer_most_expr: {outer_most_expr}")
                dyn_1 = DynDim(
                    Piecewise(
                        (Integer(1), outer_most_expr >= 1),
                        (Integer(0), outer_most_expr < 1),
                    )
                )
                # print(f"dyn_1: {dyn_1}")

                self._stream = Stream(
                    stream_dtype=in_stream.stream_dtype,
                    shape=(dyn_1,)
                    + in_stream.shape[:rank_pos]
                    + ((in_stream.shape[rank_pos] + chunk_size - 1) // chunk_size,)
                    + (chunk_size,)
                    + in_stream.shape[(rank_pos + 1) :],
                )

            else:
                self._stream = Stream(
                    stream_dtype=in_stream.stream_dtype,
                    shape=in_stream.shape[:rank_pos]
                    + ((in_stream.shape[rank_pos] + chunk_size - 1) // chunk_size,)
                    + (chunk_size,)
                    + in_stream.shape[(rank_pos + 1) :],
                )
        elif isinstance(in_stream.shape[rank_pos], int):
            if add_outer_dim:  # this means in_stream.rank == 0
                assert (
                    in_stream.shape[rank_pos] > 0
                ), "The outer most dimension must be greater than 0 if add_outer_dim is True for static case"
                self._stream = Stream(
                    stream_dtype=in_stream.stream_dtype,
                    shape=(1,)
                    + in_stream.shape[:rank_pos]
                    + ((in_stream.shape[rank_pos] + chunk_size - 1) // chunk_size,)
                    + (chunk_size,)
                    + in_stream.shape[(rank_pos + 1) :],
                )

            else:
                self._stream = Stream(
                    stream_dtype=in_stream.stream_dtype,
                    shape=in_stream.shape[:rank_pos]
                    + ((in_stream.shape[rank_pos] + chunk_size - 1) // chunk_size,)
                    + (chunk_size,)
                    + in_stream.shape[(rank_pos + 1) :],
                )
        else:
            raise ValueError(
                f"Unsupported shape type at rank {rank_pos}: {in_stream.shape[rank_pos]} (type: {type(in_stream.shape[rank_pos])})"
            )

        input_node = input if isinstance(input, StepOps) else input[0]
        graph.add_edge(input_node, self)

    @property
    def stream(self) -> Stream:
        """The stream of the operation."""
        return self._stream

    @property
    def stream_list(self) -> List[Stream]:
        return [self._stream]

    @property
    def input(self) -> Union["StepOps", Tuple["StepOps", int]]:
        return self._input

    @property
    def input_list(self) -> List[Union["StepOps", Tuple["StepOps", int]]]:
        return [self._input]

    def stream_idx(self, idx: int) -> Stream:
        raise NotImplementedError(
            "Shouldn't be called for nodes that only have a single output stream"
        )

    def __str__(self):
        cls = self.__class__.__name__
        return f"{cls}_{self.instance_id} ({self.reshape_rank} D) (chunk: {self.chunk_size})"

    def replace_input(
        self,
        org_input: Union["StepOps", Tuple["StepOps", int]],
        new_input: Union["StepOps", Tuple["StepOps", int]],
    ):
        if get_stream(self.input) != get_stream(new_input):
            raise ValueError("The shape of the input stream shouldn't change")
        self._input = new_input

    def off_chip_traffic(self) -> sympy.Expr:
        """Return the off-chip traffic for this operation."""
        return sympy.Integer(0)

    def on_chip_requirement(self, count_fifos: bool = False) -> sympy.Expr:
        """Return the on-chip memory requirement for this operation."""
        if not count_fifos:
            return sympy.Integer(0)

        # Get the size of the stream's data type
        stream_size = self._stream.stream_dtype.size_in_bytes()
        return sympy.Mul(stream_size, sympy.Integer(2))

    def T_fire(self, hw_config: dict) -> sympy.Expr:
        return sympy.Integer(0)

    def N_fire(self) -> sympy.Expr:
        return self._stream.total_elements()

    def NIT(self) -> dict:
        input_node = self._input if isinstance(self._input, StepOps) else self._input[0]
        return {input_node.instance_id: sympy.Integer(1)}

    def OTPC(self) -> sympy.Expr:
        return sympy.Integer(1)
```

## Section 3: STeP Functional Emulator (functional.py)

This is how each STeP node actually executes. Use this to understand the semantics when translating.

```python
"""Functional (value-level) simulation for STeP dataflow graphs.

Walks the graph in topological order, computing tensor values at each node.
Operates on torch.Tensors. Each node produces a tiled tensor with shape
(*stream_shape, tile_row, tile_col), where the last two dims are tile dims.

Multi-output nodes (Broadcast, FlatPartition) store a list of tensors.
"""

import torch
import torch.nn.functional as F
from networkx import topological_sort

from step_py.ops import (
    StepOps, get_stream,
    LinearOffChipLoad, LinearOffChipLoadRef,
    RandomOffChipLoad, RandomOffChipStore,
    OffChipStore, DynOffChipStore,
    BinaryMap, UnaryMap, BinaryMapAccum, Accum,
    Bufferize, Streamify, DynStreamify,
    Broadcast, RepeatStatic,
    Flatten, Reshape, ReshapePadStream, RetileStreamify,
    FlatPartition, FlatReassemble,
    ExpandRef, RepeatRef, Promote, PromoteOuter,
    MockStreamOp,
    Parallelize, StaticReassemble,
    DynLinearOffChipLoad,
    EagerMerge,
    FlatmapFilterRowStreamify,
    FlatmapCounter,
)
from step_py.utility_ops import SelectGen, MetadataGen, ExpertAddrGen
from step_py.utility_ops import CacheReadAddrGen, FilterLastTile
from step_py.utility_ops import PrinterContext, ConsumerContext
from step_py.functions import map_fn, map_accum_fn, accum_fn
from step_py.datatype import Tile, DynTile, Buffer
from step_py.ragged import RaggedTensor, unwrap, is_ragged, ragged_wrap, apply_ragged_mask, reduce_lengths


# ---------------------------------------------------------------------------
# Public API
# ---------------------------------------------------------------------------

def execute(graph, output_op, input_tensors=None):
    """Execute *graph* functionally and return the untiled output tensor.

    input_tensors: optional dict mapping tensor names to torch.Tensors,
                   used by DynLinearOffChipLoad nodes.
    """
    if input_tensors is None:
        input_tensors = {}
    values = {}

    for node in topological_sort(graph):
        values[node.instance_id] = _dispatch(node, values, input_tensors)

    return _untile_store(output_op, values)


def execute_values(graph):
    """Execute graph and return per-node values dict.

    Returns dict mapping instance_id -> tensor (or list of tensors for
    multi-output nodes like Broadcast, FlatPartition, EagerMerge).
    """
    values = {}
    for node in topological_sort(graph):
        values[node.instance_id] = _dispatch(node, values)
    return values


# ---------------------------------------------------------------------------
# Input helpers
# ---------------------------------------------------------------------------

def _val(inp, values):
    """Resolve an input reference to its tensor value."""
    if isinstance(inp, StepOps):
        return values[inp.instance_id]
    node, idx = inp
    v = values[node.instance_id]
    assert isinstance(v, list), f"Expected multi-output for {node}, got {type(v)}"
    return v[idx]


# ---------------------------------------------------------------------------
# Tile materialization (shared by LinearOffChipLoad / LinearOffChipLoadRef)
# ---------------------------------------------------------------------------

def _materialize_tiles(underlying, stride, out_shape_tiled, tile_r, tile_c, transposed):
    """Build a tiled tensor (*out_shape_tiled, tile_r, tile_c) from *underlying*."""
    R, C = underlying.shape[-2], underlying.shape[-1]
    grid_r, grid_c = R // tile_r, C // tile_c
    batch_shape = underlying.shape[:-2]

    # Slice to evenly divisible size (truncate trailing elements)
    used_R, used_C = grid_r * tile_r, grid_c * tile_c
    if used_R != R or used_C != C:
        underlying = underlying[..., :used_R, :used_C]

    # Reshape last two dims into tile grid, keeping any leading batch dims
    tiled = underlying.reshape(*batch_shape, grid_r, tile_r, grid_c, tile_c)
    # Permute so tile spatial dims are last: (*batch, grid_r, grid_c, tile_r, tile_c)
    ndim = tiled.ndim
    perm = list(range(len(batch_shape))) + [ndim-4, ndim-2, ndim-3, ndim-1]
    tiled = tiled.permute(*perm)
    # Flatten batch + grid dims into a single tile index
    flat = tiled.reshape(-1, tile_r, tile_c)

    # Compute linear tile index for every position in out_shape_tiled
    ranges = [torch.arange(s) for s in out_shape_tiled]
    grids = torch.meshgrid(*ranges, indexing="ij")
    linear_idx = sum(g.long() * int(s) for g, s in zip(grids, stride))

    result = flat[linear_idx.long()]  # (*out_shape_tiled, tile_r, tile_c)
    if transposed:
        result = result.transpose(-2, -1)
    return result


# ---------------------------------------------------------------------------
# Dispatch
# ---------------------------------------------------------------------------

def _dispatch(node, values, input_tensors=None):
    # Source nodes
    if isinstance(node, DynLinearOffChipLoad):
        return _exec_dyn_linear_load(node, input_tensors or {})

    if isinstance(node, LinearOffChipLoad):
        tiles = _materialize_tiles(
            node.underlying, node.stride, node.out_shape_tiled,
            node.tile_row, node.tile_col, node.transposed,
        )
        return tiles.unsqueeze(0)  # prepend leading 1

    if isinstance(node, LinearOffChipLoadRef):
        return _exec_load_ref(node, values)

    if isinstance(node, RandomOffChipLoad):
        return _exec_random_load(node, values)

    if isinstance(node, RandomOffChipStore):
        return _val(node.wdata, values)

    if isinstance(node, SelectGen):
        return node.underlying  # raw selection tensor

    if isinstance(node, MetadataGen):
        # MetadataGen carries actual data in .underlying (like SelectGen)
        # Stream shape = (1,) + tensor.shape, tile = (1,1)
        return node.underlying.float().reshape(*node.stream.shape, 1, 1)

    if isinstance(node, ExpertAddrGen):
        # ExpertAddrGen — address computation, produce zeros with correct shape
        stream = node.stream
        tile_shape = stream.stream_dtype.shape if hasattr(stream.stream_dtype, 'shape') else (1, 1)
        return torch.zeros(*stream.shape, *tile_shape)

    if isinstance(node, CacheReadAddrGen):
        return _exec_cache_read_addr_gen(node, values)

    if isinstance(node, FilterLastTile):
        return _exec_filter_last_tile(node, values)

    if isinstance(node, MockStreamOp):
        stream = node.stream
        tile_r, tile_c = stream.stream_dtype.shape
        shape = tuple(stream.shape) + (tile_r, tile_c)
        return torch.zeros(shape)

    # Sink / observer nodes
    if isinstance(node, (OffChipStore, DynOffChipStore)):
        return unwrap(_val(node._input, values))

    if isinstance(node, (PrinterContext, ConsumerContext)):
        return None  # graph sinks with no output

    # Compute nodes
    if isinstance(node, UnaryMap):
        inp = _val(node._input, values)
        result = _apply_unary(node.fn, unwrap(inp))
        return ragged_wrap(result, inp)

    if isinstance(node, BinaryMap):
        a_val = _val(node.in1, values)
        b_val = _val(node.in2, values)
        result = _apply_binary(node.fn, unwrap(a_val), unwrap(b_val))
        ref = a_val if is_ragged(a_val) else b_val
        return ragged_wrap(result, ref)

    if isinstance(node, BinaryMapAccum):
        return _exec_binary_map_accum(node, values)

    if isinstance(node, Accum):
        return _exec_accum(node, values)

    # Structural nodes
    if isinstance(node, Broadcast):
        inp = _val(node._input, values)
        if is_ragged(inp):
            return [RaggedTensor(inp.data.clone(), inp.lengths.copy()) for _ in range(node.num_consumers)]
        return [inp] * node.num_consumers

    if isinstance(node, Parallelize):
        inp = _val(node._input, values)
        n = node.num_consumers
        chunks = torch.chunk(inp, n, dim=0)
        return list(chunks)

    if isinstance(node, StaticReassemble):
        parts = [_val(inp, values) for inp in node._inputs]
        result = torch.cat(parts, dim=0)
        # When inputs come from Parallelize, each chunk retains a leading
        # singleton dim.  After cat the stream dims may be split (e.g.
        # (4, 16, 8) instead of (64, 8)).  Reshape to the IR output stream
        # shape so downstream ops see the correct dimensionality.
        tile_r, tile_c = result.shape[-2], result.shape[-1]
        target = tuple(node.stream.shape) + (tile_r, tile_c)
        if result.shape != target:
            result = result.reshape(target)
        return result

    if isinstance(node, RepeatStatic):
        return _exec_repeat_static(node, values)

    if isinstance(node, Bufferize):
        return _val(node._input, values)  # identity on data

    if isinstance(node, Streamify):
        return _exec_streamify(node, values)

    if isinstance(node, DynStreamify):
        return _exec_dyn_streamify(node, values)

    if isinstance(node, Flatten):
        return _exec_flatten(node, values)

    if isinstance(node, Reshape):
        return _exec_reshape(node, values)

    if isinstance(node, ReshapePadStream):
        return _exec_reshape_pad_stream(node, values)

    if isinstance(node, RetileStreamify):
        return _exec_retile_streamify(node, values)

    if isinstance(node, ExpandRef):
        return _exec_expand_ref(node, values)

    if isinstance(node, RepeatRef):
        return _exec_repeat_ref(node, values)

    if isinstance(node, Promote):
        return _exec_promote(node, values)

    if isinstance(node, PromoteOuter):
        raw_inp = _val(node._input, values)
        inp = unwrap(raw_inp)
        result = inp.unsqueeze(0)
        return ragged_wrap(result, raw_inp)

    # Routing nodes
    if isinstance(node, EagerMerge):
        return _exec_eager_merge(node, values)

    if isinstance(node, FlatPartition):
        return _exec_flat_partition(node, values)

    if isinstance(node, FlatReassemble):
        return _exec_flat_reassemble(node, values)

    if isinstance(node, FlatmapFilterRowStreamify):
        return _exec_flatmap_filter_row(node, values)

    if isinstance(node, FlatmapCounter):
        return _exec_flatmap_counter(node, values)

    assert False, f"Unsupported op: {type(node).__name__} ({node})"

def _exec_dyn_linear_load(node, input_tensors):
    name = node.input_tensor_name
    assert name in input_tensors, (
        f"DynLinearOffChipLoad requires tensor '{name}' in input_tensors"
    )
    underlying = input_tensors[name]
    tile_r, tile_c = node.tile_row, node.tile_col
    R, C = underlying.shape[-2], underlying.shape[-1]
    grid_r, grid_c = R // tile_r, C // tile_c

    tiled = underlying.reshape(grid_r, tile_r, grid_c, tile_c).permute(0, 2, 1, 3)
    # Reshape to match tensor_shape_tiled + tile dims
    out_shape = tuple(
        int(d) if not hasattr(d, 'expr') else int(d.expr)
        for d in node.tensor_shape_tiled
    )
    result = tiled.reshape(*out_shape, tile_r, tile_c)
    return result.unsqueeze(0)  # prepend leading 1


def _exec_load_ref(node, values):
    """LinearOffChipLoadRef: load weight tiles, broadcast over ref dims."""
    tiles = _materialize_tiles(
        node.underlying, node.stride, node.out_shape_tiled,
        node.tile_row, node.tile_col, node.transposed,
    )
    # tiles: (*out_shape_tiled, tile_r, tile_c)

    ref_val = _val(node.ref, values)
    ref_data = unwrap(ref_val)
    ref_stream_shape = ref_data.shape[:-2]
    if node.trigger_rank > 0:
        outer = ref_stream_shape[:-node.trigger_rank]
    else:
        outer = ref_stream_shape

    # Broadcast tiles over outer dims
    for _ in range(len(outer)):
        tiles = tiles.unsqueeze(0)
    expand = list(outer) + list(tiles.shape[len(outer):])
    return tiles.expand(expand).contiguous()


def _exec_random_load(node, values):
    """RandomOffChipLoad: index into a tiled tensor using an address stream."""
    underlying = node.underlying
    tile_r, tile_c = node.tile_row, node.tile_col
    R, C = underlying.shape[-2], underlying.shape[-1]
    grid_r, grid_c = R // tile_r, C // tile_c

    # Build flat tile array: (grid_r * grid_c, tile_r, tile_c)
    tiled = underlying.reshape(grid_r, tile_r, grid_c, tile_c).permute(0, 2, 1, 3)
    flat_tiles = tiled.reshape(-1, tile_r, tile_c)

    # addr stream shape: (*stream_shape, 1, 1) — extract scalar index from each tile
    addr_raw = _val(node.raddr, values)
    addr_data = unwrap(addr_raw)
    addr_indices = addr_data[..., 0, 0].long()

    result = flat_tiles[addr_indices]

    if node.transposed:
        result = result.transpose(-2, -1)

    # If address stream is ragged, zero out padding tiles and propagate lengths
    if is_ragged(addr_raw):
        lengths = addr_raw.lengths.copy()
        rt = RaggedTensor(result, lengths)
        for dim in list(lengths.keys()):
            result = apply_ragged_mask(rt, dim, identity=0.0)
            rt = RaggedTensor(result, lengths)
        return RaggedTensor(result, lengths)

    return result


def _exec_cache_read_addr_gen(node, values):
    """CacheReadAddrGen: compute KV cache tile addresses.

    For each (batch, head) element with sequence length N, generates
    linear tile addresses: idx * row_offset + col for col in 0..N-1.
    Output shape: (*idx_stream_shape, DynN, 1, 1)
    """
    idx = _val(node.idx, values)        # (*stream, 1, 1) — batch indices
    seq_len = _val(node.seq_len, values) # (*stream, 1, 1) — sequence lengths
    row_offset = node.row_offset

    # Extract scalar values
    idx_vals = idx[..., 0, 0]       # (*stream)
    seq_vals = seq_len[..., 0, 0]   # (*stream)

    stream_shape = idx_vals.shape
    flat_idx = idx_vals.reshape(-1)
    flat_seq = seq_vals.reshape(-1).long()

    # Maximum sequence length determines the output's DynDim
    max_seq = flat_seq.max().item()

    # Build address tensor: for each element, addr[col] = idx * row_offset + col
    addrs = []
    for i in range(flat_idx.shape[0]):
        n = flat_seq[i].item()
        base = flat_idx[i] * row_offset
        cols = base + torch.arange(n, dtype=idx.dtype)
        # Pad to max_seq with zeros (won't be accessed for shorter sequences)
        if n < max_seq:
            cols = torch.cat([cols, torch.zeros(max_seq - n, dtype=idx.dtype)])
        addrs.append(cols)

    # Stack: (flat_batch, max_seq) -> reshape to (*stream_shape, max_seq, 1, 1)
    result = torch.stack(addrs, dim=0)
    data = result.reshape(*stream_shape, max_seq, 1, 1)

    # The new max_seq dim is ragged — its position is len(stream_shape)
    ragged_dim = len(stream_shape)
    lengths_tensor = flat_seq.reshape(stream_shape)
    return RaggedTensor(data, {ragged_dim: lengths_tensor})


def _exec_filter_last_tile(node, values):
    """FilterLastTile: produce a multihot marking last vs non-last tiles.

    Input is a sequence length stream (tile: 1x1). For each element with
    value N, produces N multihot entries: [1,0] for positions 0..N-2 (non-last),
    [0,1] for position N-1 (last).
    Output shape: (*input_stream, DynN, 2) — a multihot with 2 categories.
    """
    inp = _val(node._input, values)
    seq_vals = inp[..., 0, 0]  # (*stream) — sequence lengths

    stream_shape = seq_vals.shape
    flat_seq = seq_vals.reshape(-1).long()
    max_seq = flat_seq.max().item()

    # Build multihot: column 0 = non-last, column 1 = last
    entries = []
    for i in range(flat_seq.shape[0]):
        n = flat_seq[i].item()
        mh = torch.zeros(max_seq, 2)
        if n > 0:
            mh[:n-1, 0] = 1.0   # non-last tiles
            mh[n-1, 1] = 1.0    # last tile
        entries.append(mh)

    result = torch.stack(entries, dim=0)
    data = result.reshape(*stream_shape, max_seq, 2)

    # The new max_seq dim is ragged — its position is len(stream_shape)
    ragged_dim = len(stream_shape)
    lengths_tensor = flat_seq.reshape(stream_shape)
    return RaggedTensor(data, {ragged_dim: lengths_tensor})

def _apply_unary(fn, x):
    if isinstance(fn, map_fn.Square):
        return x ** 2
    if isinstance(fn, map_fn.Silu):
        return F.silu(x)
    if isinstance(fn, map_fn.Exp):
        return torch.exp(x)
    if isinstance(fn, map_fn.Rsqrt):
        return torch.rsqrt(x)
    if isinstance(fn, map_fn.Pow2):
        return torch.pow(2.0, x)
    if isinstance(fn, map_fn.MulImmediate):
        return x * fn.constant
    if isinstance(fn, map_fn.AddImmediate):
        return x + fn.constant
    if isinstance(fn, map_fn.SubImmediate):
        return x - fn.constant
    if isinstance(fn, map_fn.RowWiseSum):
        return x.sum(dim=-1, keepdim=True)
    if isinstance(fn, map_fn.MaskRow):
        return torch.ones_like(x[..., :1])
    if isinstance(fn, map_fn.SelectToScalar):
        return x
    if isinstance(fn, map_fn.ToConstInt):
        return torch.full_like(x, fn.constant, dtype=torch.float32)
    assert False, f"Unsupported unary fn: {type(fn).__name__}"


def _apply_binary(fn, a, b):
    if isinstance(fn, (map_fn.Matmul, map_fn.DynMatmul)):
        if fn.weight_transposed:
            return torch.matmul(a, b.transpose(-2, -1))
        return torch.matmul(a, b)
    if isinstance(fn, (map_fn.Add,)):
        return a + b
    if isinstance(fn, (map_fn.Mul,)):
        return a * b
    if isinstance(fn, (map_fn.Div,)):
        return a / b
    if isinstance(fn, map_fn.IsEqual):
        return (a == b).float()
    if isinstance(fn, map_fn.SetOffset):
        return a  # pass-through: offset metadata not needed for value correctness
    if isinstance(fn, map_fn.RowWiseAppend):
        return a  # pass-through: append semantics handled by surrounding graph structure
    if isinstance(fn, map_fn.CacheWriteAddrGen):
        return a  # address generation — not needed for data path
    assert False, f"Unsupported binary fn: {type(fn).__name__}"

def _exec_binary_map_accum(node, values):
    a_raw = _val(node.in1, values)
    b_raw = _val(node.in2, values)
    a = unwrap(a_raw)
    b = unwrap(b_raw)
    fn = node.fn

    # Inherit ragged metadata from whichever input is ragged
    ref = a_raw if is_ragged(a_raw) else b_raw
    lengths = ref.lengths.copy() if is_ragged(ref) else {}

    # Apply per-tile map function (batched)
    if isinstance(fn, (map_accum_fn.Matmul, map_accum_fn.DynMatmul)):
        if fn.weight_transposed:
            mapped = torch.matmul(a, b.transpose(-2, -1))
        else:
            mapped = torch.matmul(a, b)
    else:
        assert False, f"Unsupported map_accum fn: {type(fn).__name__}"

    # Sum-reduce over the last `rank` stream dims (sitting at dim -3 after tile dims)
    for _ in range(node.rank):
        reduce_dim = mapped.ndim - 3
        if reduce_dim in lengths:
            rt = RaggedTensor(mapped, lengths)
            mapped = apply_ragged_mask(rt, reduce_dim, identity=0.0)
            lengths = reduce_lengths(lengths, reduce_dim)
        mapped = mapped.sum(dim=-3)
    return RaggedTensor(mapped, lengths) if lengths else mapped


def _exec_accum(node, values):
    raw_inp = _val(node._input, values)
    fn = node.fn
    rank = node.accum_rank

    if isinstance(fn, accum_fn.Add):
        inp = raw_inp
        lengths = inp.lengths.copy() if is_ragged(inp) else {}
        data = unwrap(inp)
        for _ in range(rank):
            reduce_dim = data.ndim - 3
            if reduce_dim in lengths:
                rt = RaggedTensor(data, lengths)
                data = apply_ragged_mask(rt, reduce_dim, identity=0.0)
                lengths = reduce_lengths(lengths, reduce_dim)
            data = data.sum(dim=-3)
        return RaggedTensor(data, lengths) if lengths else data

    if isinstance(fn, accum_fn.Mul):
        inp = raw_inp
        lengths = inp.lengths.copy() if is_ragged(inp) else {}
        data = unwrap(inp)
        for _ in range(rank):
            reduce_dim = data.ndim - 3
            if reduce_dim in lengths:
                rt = RaggedTensor(data, lengths)
                data = apply_ragged_mask(rt, reduce_dim, identity=1.0)
                lengths = reduce_lengths(lengths, reduce_dim)
            data = data.prod(dim=-3)
        return RaggedTensor(data, lengths) if lengths else data

    if isinstance(fn, accum_fn.RetileRow):
        inp = unwrap(raw_inp)
        for _ in range(rank):
            s = inp.shape
            inp = inp.reshape(*s[:-3], s[-3] * s[-2], s[-1])
        return inp

    if isinstance(fn, accum_fn.RetileCol):
        inp = unwrap(raw_inp)
        for _ in range(rank):
            s = inp.shape
            inp = inp.reshape(*s[:-3], s[-2], s[-3] * s[-1])
        return inp

    if isinstance(fn, accum_fn.SignalReqAllRead):
        inp = unwrap(raw_inp)
        stream_shape = inp.shape[:-2 - rank]
        return torch.ones(*stream_shape, 1, 1)

    assert False, f"Unsupported accum fn: {type(fn).__name__}"

def _exec_repeat_static(node, values):
    raw_inp = _val(node._input, values)
    inp = unwrap(raw_inp)
    # Insert new dim at position -3 (before tile dims) and expand
    result = inp.unsqueeze(-3)
    shape = list(result.shape)
    shape[-3] = node.repeat_factor
    result = result.expand(shape).contiguous()
    return ragged_wrap(result, raw_inp)


def _exec_streamify(node, values):
    inp = unwrap(_val(node._input, values))
    if not node.repeat_factor:
        return inp  # identity

    buf_rank = node.rank
    result = inp
    offset = 2 + buf_rank  # skip buffer_dims + tile_r + tile_c
    for rf in node.repeat_factor:
        result = result.unsqueeze(-offset)
        shape = list(result.shape)
        shape[-offset] = rf
        result = result.expand(shape).contiguous()
        offset += 1  # account for the newly inserted dim
    return result


def _exec_dyn_streamify(node, values):
    inp = unwrap(_val(node._input, values))
    ref = unwrap(_val(node.ref, values))
    ref_stream_shape = ref.shape[:-2]
    buf_and_tile_dims = inp.shape[-(2 + node.bufferized_rank):]
    expand_shape = list(ref_stream_shape) + list(buf_and_tile_dims)
    return inp.expand(expand_shape).contiguous()


def _exec_flatten(node, values):
    inp = unwrap(_val(node._input, values))
    tile_r, tile_c = inp.shape[-2], inp.shape[-1]
    stream_shape = inp.shape[:-2]

    # Flatten merges stream dimensions between min_rank and max_rank.
    # Ranks count from the right (rank 0 = rightmost stream dim).
    min_rank = node.min_rank
    max_rank = node.max_rank
    n = len(stream_shape)
    min_idx = n - 1 - max_rank  # max_rank -> leftmost merged index
    max_idx = n - 1 - min_rank  # min_rank -> rightmost merged index

    merged = 1
    for i in range(min_idx, max_idx + 1):
        merged *= stream_shape[i]

    new_stream = list(stream_shape[:min_idx]) + [merged] + list(stream_shape[max_idx + 1:])
    return inp.reshape(*new_stream, tile_r, tile_c)


def _exec_reshape(node, values):
    inp = unwrap(_val(node._input, values))
    tile_r, tile_c = inp.shape[-2], inp.shape[-1]
    stream_dims = list(inp.shape[:-2])

    chunk = node.chunk_size
    rank_pos = node.input_stream_rank - node.reshape_rank

    D = stream_dims[rank_pos]
    padded_D = ((D + chunk - 1) // chunk) * chunk

    if padded_D != D:
        # Pad along the rank_pos dimension
        pad_sizes = [0] * (2 * len(inp.shape))
        pad_idx = 2 * (len(inp.shape) - 1 - rank_pos)
        pad_sizes[pad_idx + 1] = padded_D - D
        inp = torch.nn.functional.pad(inp, pad_sizes, value=0.0)
        stream_dims[rank_pos] = padded_D

    pre = stream_dims[:rank_pos]
    post = stream_dims[rank_pos + 1:]
    new_count = padded_D // chunk

    if node.add_outer_dim:
        new_shape = [1] + pre + [new_count, chunk] + post + [tile_r, tile_c]
    else:
        new_shape = pre + [new_count, chunk] + post + [tile_r, tile_c]

    return inp.reshape(new_shape)


def _exec_reshape_pad_stream(node, values):
    inp = unwrap(_val(node._input, values))
    tile_r, tile_c = inp.shape[-2], inp.shape[-1]
    stream_dims = list(inp.shape[:-2])

    chunk = node.chunk_size
    rank_pos = node.input_stream_rank - node.reshape_rank

    D = stream_dims[rank_pos]
    padded_D = ((D + chunk - 1) // chunk) * chunk

    # Build pad mask before potentially modifying stream_dims
    pad_count = padded_D - D

    if pad_count != 0:
        pad_sizes = [0] * (2 * len(inp.shape))
        pad_idx = 2 * (len(inp.shape) - 1 - rank_pos)
        pad_sizes[pad_idx + 1] = pad_count
        inp = F.pad(inp, pad_sizes, value=0.0)
        stream_dims[rank_pos] = padded_D

    pre = stream_dims[:rank_pos]
    post = stream_dims[rank_pos + 1:]
    new_count = padded_D // chunk
    new_shape = pre + [new_count, chunk] + post + [tile_r, tile_c]
    data = inp.reshape(new_shape)

    # Build pad mask: 1.0 for real tiles, 0.0 for padded tiles
    # Mask has same stream shape as data but tile dims (1, 1)
    mask_stream_shape = pre + [new_count, chunk] + post
    total = 1
    for s in mask_stream_shape:
        total *= s
    real_count = D  # number of real tiles along the split dimension
    # Flatten the split dim (rank_pos) and mark the last pad_count entries as 0
    mask_flat = torch.ones(padded_D, dtype=torch.float32)
    if pad_count != 0:
        mask_flat[D:] = 0.0
    # Reshape mask_flat to match new stream shape along the split dims, then broadcast
    split_shape = pre + [new_count, chunk] + post
    # Construct full mask tensor over all stream dims
    mask = mask_flat.reshape([1] * len(pre) + [new_count, chunk] + [1] * len(post))
    mask = mask.expand(split_shape).contiguous()
    # Add tile dims (1, 1)
    mask = mask.unsqueeze(-1).unsqueeze(-1)

    if node.have_pad_stream:
        return [data, mask]
    return data


def _exec_retile_streamify(node, values):
    inp = unwrap(_val(node._input, values))
    chunk = node.chunk
    num_chunks = node.num_chunks

    if node.split_row:
        # (..., last_dim, tile_r, tile_c) -> (..., last_dim, num_chunks, chunk, tile_c)
        batch = inp.shape[:-2]
        last = batch[-1]
        pre = batch[:-1]
        tile_r, tile_c = inp.shape[-2], inp.shape[-1]

        # Compute actual num_chunks from tensor (may differ from IR after RetileRow merge)
        actual_num_chunks = tile_r // chunk
        reshaped = inp.reshape(*pre, last, actual_num_chunks, chunk, tile_c)
        result = reshaped.reshape(*pre, last * actual_num_chunks, chunk, tile_c)

        if node.filter_mask:
            # Keep only non-zero rows (remove padding)
            flat = result.reshape(-1, chunk, tile_c)
            mask = flat.abs().sum(dim=(-2, -1)) > 0
            result = flat[mask]  # (valid_count, chunk, tile_c)
        return result

    # split_col
    batch = inp.shape[:-2]
    last = batch[-1]
    pre = batch[:-1]
    tile_r, tile_c = inp.shape[-2], inp.shape[-1]

    # Compute actual num_chunks from tensor dimensions (may differ from IR
    # metadata if an upstream Accum(RetileCol) merged tile columns).
    actual_num_chunks = tile_c // chunk
    reshaped = inp.reshape(*pre, last, tile_r, actual_num_chunks, chunk)
    # Rearrange so chunks come before tile_r
    perm = list(range(len(pre))) + [len(pre), len(pre) + 2, len(pre) + 1, len(pre) + 3]
    reshaped = reshaped.permute(perm)
    result = reshaped.reshape(*pre, last * actual_num_chunks, tile_r, chunk)

    if node.filter_mask:
        flat = result.reshape(-1, tile_r, chunk)
        mask = flat.abs().sum(dim=(-2, -1)) > 0
        result = flat[mask]
    return result


def _exec_expand_ref(node, values):
    raw_inp = _val(node._input, values)
    raw_ref = _val(node.ref, values)
    inp = unwrap(raw_inp)
    ref = unwrap(raw_ref)
    # Expand singleton stream dims to match ref stream dims
    ref_shape = ref.shape[:-2]
    expand_shape = list(ref_shape) + list(inp.shape[-2:])
    result = inp.expand(expand_shape).contiguous()
    # Inherit raggedness from ref (the reference stream defines the ragged structure)
    return ragged_wrap(result, raw_ref)


def _exec_repeat_ref(node, values):
    raw_inp = _val(node._input, values)
    raw_ref = _val(node.ref, values)
    inp = unwrap(raw_inp)
    ref = unwrap(raw_ref)
    ref_stream = list(ref.shape[:-2])
    inp_stream = list(inp.shape[:-2])
    tile_dims = list(inp.shape[-2:])

    # Align inp stream dims with ref stream dims by inserting singletons
    # where ref has dims that inp doesn't.  Work from the right (tile end).
    aligned = tile_dims[:]
    ri = len(ref_stream) - 1
    ii = len(inp_stream) - 1
    while ri >= 0:
        if ii >= 0 and inp_stream[ii] == ref_stream[ri]:
            aligned.insert(0, inp_stream[ii])
            ri -= 1
            ii -= 1
        elif ii >= 0 and inp_stream[ii] == 1:
            aligned.insert(0, 1)
            ri -= 1
            ii -= 1
        else:
            aligned.insert(0, 1)
            ri -= 1

    result = inp.reshape(aligned)
    expand_shape = ref_stream + tile_dims
    result = result.expand(expand_shape).contiguous()
    # Inherit raggedness from ref (the reference stream defines the ragged structure)
    return ragged_wrap(result, raw_ref)


def _exec_promote(node, values):
    raw_inp = _val(node._input, values)
    inp = unwrap(raw_inp)
    # Promote inserts a singleton dim at rank (promote_rank) from the right of stream dims.
    # In our tensor, stream dims are everything except last 2 (tile dims).
    promote_rank = node.promote_rank
    # position = -(2 + promote_rank)
    result = inp.unsqueeze(-(2 + promote_rank))
    return ragged_wrap(result, raw_inp)


# ---------------------------------------------------------------------------
# Routing ops (EagerMerge / FlatPartition / FlatReassemble)
# ---------------------------------------------------------------------------

def _exec_eager_merge(node, values):
    parts = [unwrap(_val(inp, values)) for inp in node._inputs]
    n = len(parts)

    # Concatenate data along outermost dim
    data = torch.cat(parts, dim=0)

    # Build selection multihot: for each tile, a one-hot indicating which input
    counts = [p.shape[0] for p in parts]
    total = sum(counts)
    select = torch.zeros(total, n)
    offset = 0
    for i, c in enumerate(counts):
        select[offset:offset + c, i] = 1.0
        offset += c

    return [data, select]


def _exec_flat_partition(node, values):
    inp = unwrap(_val(node._input, values))
    multihot = unwrap(_val(node.control, values))
    n = node.num_consumers

    tile_r, tile_c = inp.shape[-2], inp.shape[-1]
    flat_inp = inp.reshape(-1, tile_r, tile_c)
    flat_mh = multihot.reshape(-1, n)

    assert flat_inp.shape[0] == flat_mh.shape[0], (
        f"Tile count mismatch: {flat_inp.shape[0]} vs {flat_mh.shape[0]}"
    )

    results = []
    for i in range(n):
        mask = flat_mh[:, i] > 0
        results.append(flat_inp[mask])
    return results


def _exec_flat_reassemble(node, values):
    inputs = [unwrap(_val(inp, values)) for inp in node._inputs]
    multihot = unwrap(_val(node.control, values))
    n = len(inputs)

    tile_r, tile_c = inputs[0].shape[-2], inputs[0].shape[-1]
    flat_mh = multihot.reshape(-1, n)
    total = flat_mh.shape[0]

    # Track read pointer per expert stream
    ptrs = [0] * n
    token_groups = []

    for t in range(total):
        group = []
        for i in range(n):
            if flat_mh[t, i] > 0:
                if ptrs[i] < inputs[i].shape[0]:
                    group.append(inputs[i][ptrs[i]])
                    ptrs[i] += 1
        if len(group) == 0:
            # Token matched no expert — emit a zero tile
            group.append(torch.zeros_like(inputs[0][0:1].squeeze(0)))
        token_groups.append(torch.stack(group, dim=0))

    # Stack: (total, n_active, tile_r, tile_c)
    output = torch.stack(token_groups, dim=0)

    # Reshape to match control stream shape with leading 1
    ctrl_stream_shape = multihot.shape[:-1]
    if ctrl_stream_shape[0] != 1:
        ctrl_stream_shape = (1,) + ctrl_stream_shape
    n_active = output.shape[1]
    return output.reshape(*ctrl_stream_shape, n_active, tile_r, tile_c)


def _exec_flatmap_filter_row(node, values):
    data = unwrap(_val(node._input, values))
    mask = unwrap(_val(node.mask, values))

    tile_r, tile_c = data.shape[-2], data.shape[-1]
    # Flatten all stream dims into a batch of tiles
    flat_data = data.reshape(-1, tile_r, tile_c)
    flat_mask = mask.reshape(-1, tile_r, 1)

    # Collect rows where mask > 0
    rows = []
    for i in range(flat_data.shape[0]):
        for r in range(tile_r):
            if flat_mask[i, r, 0] > 0:
                rows.append(flat_data[i, r:r+1, :])  # (1, tile_c)

    assert len(rows) > 0, "FlatmapFilterRowStreamify: no rows passed the mask"
    result = torch.cat(rows, dim=0)  # (num_selected, tile_c) with leading tile dim 1

    # Reshape: outer stream dims preserved, innermost becomes dynamic
    # data.shape = (*stream_shape, tile_r, tile_c); stream_shape[0] is the leading 1
    stream_shape = data.shape[:-2]
    outer = stream_shape[:-1]
    return result.reshape(*outer, len(rows), 1, tile_c)


def _exec_flatmap_counter(node, values):
    inp = unwrap(_val(node._input, values))
    # inp: (*stream_shape, 1, 1) — each scalar generates a count sequence
    stream_shape = inp.shape[:-2]
    flat = inp.reshape(-1)
    assert flat.numel() == 1, "FlatmapCounter with multiple scalars not yet supported"
    n = int(flat[0].item())
    return torch.arange(n, dtype=inp.dtype).reshape(*stream_shape, n, 1, 1)

def _untile_store(store_op, values):
    """Reconstruct a dense 2-D tensor from the tiled store input."""
    inp = _val(store_op._input, values)
    tile_r = store_op.tile_row
    tile_c = store_op.tile_col

    # Strip the leading 1 from the load convention if present.
    # Some graphs (e.g. end_to_end after Parallelize/StaticReassemble/Flatten)
    # may not have a leading singleton by the time data reaches the store.
    if inp.shape[0] == 1:
        inp = inp[0]

    tiled = inp.shape[:-2]

    if len(tiled) == 0:
        return inp  # single tile

    if len(tiled) == 1:
        return inp.reshape(tiled[0] * tile_r, tile_c)

    # 2-D+ tiled: last two tiled dims are row/col tile counts
    Tc = tiled[-1]
    ndim = len(inp.shape)
    perm = list(range(ndim - 4)) + [ndim - 4, ndim - 2, ndim - 3, ndim - 1]
    inp = inp.permute(*perm).contiguous()

    total_rows = tile_r
    for d in tiled[:-1]:
        total_rows *= d
    return inp.reshape(int(total_rows), int(Tc * tile_c))
```
